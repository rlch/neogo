The `MATCH` clause is the primary way to query data in Neo4j. neogo provides a fluent API for building complex match patterns.

## Basic MATCH

### Matching Nodes

```go
var person Person

cy := c.Match(
    db.Node(&person),
).Return(&person)
```

Generated Cypher:
```cypher
MATCH (person:Person)
RETURN person
```

### Matching with Properties

```go
var person Person

cy := c.Match(
    db.Node(&person, db.Props{"name": "Alice"}),
).Return(&person)
```

Generated Cypher:
```cypher
MATCH (person:Person {name: "Alice"})
RETURN person
```

## Executing Queries

### Single Result

```go
result, err := cy.Single(ctx, tx)
if err != nil {
    if errors.Is(err, neogo.ErrNotFound) {
        // No results
    }
    return err
}
fmt.Println(result.Person.Name)
```

### All Results

```go
results, err := cy.All(ctx, tx)
if err != nil {
    return err
}
for _, r := range results {
    fmt.Println(r.Person.Name)
}
```

### Streaming Results

```go
err := cy.Stream(ctx, tx, func(r *Result) error {
    fmt.Println(r.Person.Name)
    return nil
})
```

## Matching Relationships

### Basic Relationship Match

```go
var person, friend Person
var knows Knows

cy := c.Match(
    db.Patterns(
        db.Node(&person),
        db.Related(&knows, db.To, &friend),
    ),
).Return(&person, &friend)
```

Generated Cypher:
```cypher
MATCH (person:Person)-[knows:KNOWS]->(friend:Person)
RETURN person, friend
```

## Multiple MATCH Clauses

### Sequential Matches

```go
var person Person
var company Company

cy := c.Match(
    db.Node(&person, db.Props{"name": "Alice"}),
).Match(
    db.Node(&company, db.Props{"name": "Acme"}),
).Return(&person, &company)
```

Generated Cypher:
```cypher
MATCH (person:Person {name: "Alice"})
MATCH (company:Company {name: "Acme"})
RETURN person, company
```

### Multiple Patterns in One MATCH

```go
cy := c.Match(
    db.Node(&person),
    db.Node(&company),
).Return(&person, &company)
```

Generated Cypher:
```cypher
MATCH (person:Person), (company:Company)
RETURN person, company
```

## Pattern Chains

### Multi-Hop Patterns

```go
var a, b, c Person
var r1, r2 Knows

cy := c.Match(
    db.Patterns(
        db.Node(&a),
        db.Related(&r1, db.To, &b),
        db.Related(&r2, db.To, &c),
    ),
).Return(&a, &b, &c)
```

Generated Cypher:
```cypher
MATCH (a:Person)-[r1:KNOWS]->(b:Person)-[r2:KNOWS]->(c:Person)
RETURN a, b, c
```

### Branching Patterns

```go
var person Person
var friend1, friend2 Person

cy := c.Match(
    db.Patterns(
        db.Node(&person, db.Props{"name": "Alice"}),
        db.Related(db.Var("KNOWS"), db.To, &friend1),
    ),
).Match(
    db.Patterns(
        db.Node(&person),
        db.Related(db.Var("KNOWS"), db.To, &friend2),
    ),
).Where(
    db.Cond(&friend1, "", db.NotEqual, &friend2),
).Return(&friend1, &friend2)
```

## Filtering with WHERE

### Basic Conditions

```go
var person Person

cy := c.Match(
    db.Node(&person),
).Where(
    db.Cond(&person, "age", db.GreaterThan, 30),
).Return(&person)
```

Generated Cypher:
```cypher
MATCH (person:Person)
WHERE person.age > 30
RETURN person
```

### Multiple Conditions

```go
cy := c.Match(
    db.Node(&person),
).Where(
    db.And(
        db.Cond(&person, "age", db.GreaterThan, 25),
        db.Cond(&person, "active", db.Equal, true),
    ),
).Return(&person)
```

See [WHERE Clauses](/advanced/where) for comprehensive filtering options.

## Returning Data

### Return Nodes

```go
cy := c.Match(db.Node(&person)).Return(&person)
```

### Return Specific Properties

```go
cy := c.Match(db.Node(&person)).Return(
    db.Qual(&person, "name"),
    db.Qual(&person, "age"),
)
```

Generated Cypher:
```cypher
MATCH (person:Person)
RETURN person.name, person.age
```

### Return with Alias

```go
cy := c.Match(db.Node(&person)).Return(
    db.Alias(db.Qual(&person, "name"), "personName"),
)
```

### Return Distinct

```go
cy := c.Match(
    db.Patterns(
        db.Node(&person),
        db.Related(nil, db.To, db.Node(nil)),
    ),
).Return(db.Distinct(&person))
```

Generated Cypher:
```cypher
MATCH (person:Person)-->()
RETURN DISTINCT person
```

## Ordering and Pagination

### ORDER BY

```go
cy := c.Match(db.Node(&person)).
    Return(&person).
    OrderBy(db.Desc(db.Qual(&person, "age")))
```

Generated Cypher:
```cypher
MATCH (person:Person)
RETURN person
ORDER BY person.age DESC
```

### SKIP and LIMIT

```go
cy := c.Match(db.Node(&person)).
    Return(&person).
    OrderBy(db.Asc(db.Qual(&person, "name"))).
    Skip(10).
    Limit(25)
```

Generated Cypher:
```cypher
MATCH (person:Person)
RETURN person
ORDER BY person.name ASC
SKIP 10
LIMIT 25
```

## Aggregation

### Count

```go
cy := c.Match(db.Node(&person)).
    Return(db.Lit("count(person)"))
```

### Group By

```go
cy := c.Match(
    db.Patterns(
        db.Node(&person),
        db.Related(&worksAt, db.To, &company),
    ),
).Return(
    db.Qual(&company, "name"),
    db.Lit("count(person) as employee_count"),
).OrderBy(db.Desc(db.Lit("employee_count")))
```

## Anonymous Matching

### Match Any Node

```go
cy := c.Match(
    db.Node(nil),  // Any node
).Return(db.Lit("count(*)"))
```

### Match Specific Label Without Binding

```go
cy := c.Match(
    db.Node(db.Var("Person")),
).Return(db.Lit("count(*)"))
```

### Match Any Relationship

```go
cy := c.Match(
    db.Patterns(
        db.Node(&person),
        db.Related(nil, db.To, db.Node(nil)),
    ),
).Return(&person)
```

## Query Patterns

### Find Connected Nodes

```go
var person Person
var connected db.INode  // Any node type

cy := c.Match(
    db.Patterns(
        db.Node(&person, db.Props{"name": "Alice"}),
        db.Related(nil, db.Either, &connected),
    ),
).Return(&connected)
```

### Find Paths

```go
var person, target Person

cy := c.Match(
    db.Path("p",
        db.Node(&person, db.Props{"name": "Alice"}),
        db.Related(nil, db.To, &target, db.Props{"name": "Bob"}),
    ),
).Return(db.Lit("p"))
```

### Check Existence

```go
cy := c.Match(
    db.Node(&person, db.Props{"name": "Alice"}),
).Return(db.Lit("1"))

_, err := cy.Single(ctx, tx)
exists := err == nil
```
